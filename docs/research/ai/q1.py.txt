H=12
def load(p):
    d=pd.read_csv(p); d.columns=[x.lower() for x in d.columns]; d=d.rename(columns={d.columns[0]:'time'})
    d['time']=pd.to_datetime(d.time,utc=True,errors='coerce'); return d.dropna(subset=['time']).reset_index(drop=True)
def feats(d,mk,tf):
    o,h,l,c=[d[x].astype(float) for x in ['open','high','low','close']]
    pc=c.shift(); tr=pd.concat([h-l,(h-pc).abs(),(l-pc).abs()],axis=1).max(axis=1)
    a=tr.ewm(alpha=1/14,adjust=False).mean(); a5=tr.ewm(alpha=1/5,adjust=False).mean(); a50=tr.ewm(alpha=1/50,adjust=False).mean()
    am=a.rolling(100).mean()
    e20=c.ewm(span=20,adjust=False).mean(); e50=c.ewm(span=50,adjust=False).mean()
    dl=c.diff(); up=dl.clip(lower=0).ewm(alpha=1/14,adjust=False).mean(); dn=(-dl.clip(upper=0)).ewm(alpha=1/14,adjust=False).mean(); rsi=100-100/(1+up/dn)
    sd=c.rolling(20).std()
    hr=((d.time.dt.hour-(2 if mk=='gold' else 0))%24)
    F=pd.DataFrame({
        'atr_ratio':np.log(a/am),'squeeze':np.log(a5/a50),'bbw':np.log(4*sd/a),'rng1':np.log((h-l)/a+1e-3),
        'rng3':np.log((h.rolling(3).max()-l.rolling(3).min())/a+1e-3),'ret5':(c-c.shift(5))/a,'ret20':(c-c.shift(20))/a,
        'd_e20':(c-e20)/a,'d_e50':(c-e50)/a,'rsi':(rsi-50)/50,'hs':np.sin(2*np.pi*hr/24),'hc':np.cos(2*np.pi*hr/24),
        'dow':d.time.dt.dayofweek.astype(float),'tf':{'15m':0.,'1h':1.,'4h':2.}[tf],
    })
    hh=h[::-1].rolling(H).max()[::-1].shift(-1); ll=l[::-1].rolling(H).min()[::-1].shift(-1)
    F['U']=(hh-c)/a; F['D']=(c-ll)/a; F['C']=(c.shift(-H)-c)/a
    # أيهما أولاً: +1 ATR أم −1 ATR خلال H شمعة
    hv,lv,cv,av=h.values,l.values,c.values,a.values; n=len(cv); first=np.full(n,np.nan)
    for i in range(n-H-1):
        uu=cv[i]+av[i]; dd=cv[i]-av[i]
        for j in range(i+1,i+H+1):
            hu=hv[j]>=uu; hd=lv[j]<=dd
            if hu and hd: break
            if hu: first[i]=1; break
            if hd: first[i]=0; break
    F['UPF']=first
    F['yr']=d.time.dt.year; F['mk']=mk
    return F.iloc[200:-H-1]
sets=[('gold','15m','gold20/xau20_15m'),('gold','1h','gold20/xau20_1h'),('gold','4h','gold20/xau20_4h'),('btc','15m','gold10/btc10_15m'),('btc','1h','gold10/btc10_1h'),('btc','4h','gold10/btc10_4h'),('eth','1h','data/eth_1h_1819'),('eth','1h','data/eth_1h_2526'),('eur','1h','data/EUR_USD_1h')]
df=pd.concat([feats(load(f'{S}/{f}.csv'),mk,tf) for mk,tf,f in sets],ignore_index=True).replace([np.inf,-np.inf],np.nan).dropna(subset=['atr_ratio','squeeze','bbw','U','D','C'])
df.to_parquet(S+'/ai/q.parquet') if False else None
X=['atr_ratio','squeeze','bbw','rng1','rng3','ret5','ret20','d_e20','d_e50','rsi','hs','hc','dow','tf']
tr=((df.mk=='gold')&(df.yr>=2010)&(df.yr<=2018))|((df.mk=='btc')&(df.yr<=2020)); te=~tr&~((df.mk=='gold')&(df.yr<2010))
print('train',tr.sum(),'test',te.sum())
y=np.log(df.U+df.D+1e-3)
lin=LinearRegression().fit(df[tr][X],y[tr]); g=lgb.LGBMRegressor(n_estimators=300,learning_rate=0.05,num_leaves=31,min_child_samples=500,verbose=-1).fit(df[tr][X].values[::3],y[tr].values[::3])
base=y[tr].mean()
for nm,pr in [('ثابت (بدون ذكاء)',np.full(len(df),base)),('Q خطي',lin.predict(df[X])),('Q LightGBM',g.predict(df[X].values))]:
    sc=np.exp(pr)  # المدى المتوقع بوحدات ATR
    z=df.C/sc
    q=np.quantile(z[tr],[0.05,0.15,0.85,0.95])
    lo70,hi70,lo90,hi90=q[1]*sc,q[2]*sc,q[0]*sc,q[3]*sc
    out=f"{nm:18s}"
    for lab,m in [('ذهب',te&(df.mk=='gold')),('بيتكوين',te&(df.mk=='btc')),('إيثيريوم',te&(df.mk=='eth')),('يورو',te&(df.mk=='eur'))]:
        c70=((df.C>=lo70)&(df.C<=hi70))[m].mean(); c90=((df.C>=lo90)&(df.C<=hi90))[m].mean(); w70=(hi70-lo70)[m].mean()
        r2=1-((y-pr)[m]**2).mean()/((y[m]-y[m].mean())**2).mean()
        out+=f" | {lab}: 70%→{c70*100:.1f} 90%→{c90*100:.1f} عرض70 {w70:.2f}ATR دقة المدى R²={r2:.2f}"
    print(out)

        pp=p[mm]; yy=df.UPF[mm].values

        strong=(pp>=0.6)|(pp<=0.4); acc=((pp[strong]>=0.5)==(yy[strong]==1)).mean()
