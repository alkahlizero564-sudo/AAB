@njit(cache=True)
def label(o,h,l,c,a,d,w):
    n=len(c)
    for i in range(3,n-1):
        if a[i]<=0: w[i]=-1; continue
        e=o[i+1]; ext=l[i] if d>0 else h[i]
        for j in range(i-2,i+1):
            ext=min(ext,l[j]) if d>0 else max(ext,h[j])
        risk=min(max((e-ext)*d+0.1*a[i],0.7*a[i]),2.5*a[i])
        t1=e+d*risk; st=e-d*risk; res=-1
        for j in range(i+1,min(n,i+80)):
            if (d>0 and l[j]<=st) or (d<0 and h[j]>=st): res=0; break
            if (d>0 and h[j]>=t1) or (d<0 and l[j]<=t1): res=1; break
        w[i]=res if res>=0 else 0
def load(p):

@njit(cache=True)
def ctx_feats(o,h,l,c,v,a,F):
    n=len(c)
    # OB/FVG: آخر 4 مناطق لكل جهة (أعلى، أسفل، نشطة)
    obt=np.zeros((2,4)); obb=np.zeros((2,4)); oba=np.zeros((2,4),np.int64)
    fvt=np.zeros((2,4)); fvb=np.zeros((2,4)); fva=np.zeros((2,4),np.int64)
    lastH=-1; lastL=-1; hB=True; lB=True
    for i in range(160,n):
        j=i-3
        isH=True; isL=True
        for q in range(j-3,j+4):
            if q==j: continue
            if h[q]>=h[j]: isH=False
            if l[q]<=l[j]: isL=False
        if isH: lastH=j; hB=False
        if isL: lastL=j; lB=False
        if lastH>=0 and (not hB) and c[i]>h[lastH]:
            hB=True; li=lastH
            for q in range(lastH,i+1):
                if l[q]<l[li]: li=q
            for q in range(li,max(0,li-6),-1):
                if c[q]<o[q]:
                    for s_ in range(3,0,-1): obt[0,s_]=obt[0,s_-1]; obb[0,s_]=obb[0,s_-1]; oba[0,s_]=oba[0,s_-1]
                    obt[0,0]=h[q]; obb[0,0]=l[q]; oba[0,0]=1; break
        if lastL>=0 and (not lB) and c[i]<l[lastL]:
            lB=True; hi_=lastL
            for q in range(lastL,i+1):
                if h[q]>h[hi_]: hi_=q
            for q in range(hi_,max(0,hi_-6),-1):
                if c[q]>o[q]:
                    for s_ in range(3,0,-1): obt[1,s_]=obt[1,s_-1]; obb[1,s_]=obb[1,s_-1]; oba[1,s_]=oba[1,s_-1]
                    obt[1,0]=h[q]; obb[1,0]=l[q]; oba[1,0]=1; break
        if l[i]>h[i-2] and l[i]-h[i-2]>=0.15*a[i]:
            for s_ in range(3,0,-1): fvt[0,s_]=fvt[0,s_-1]; fvb[0,s_]=fvb[0,s_-1]; fva[0,s_]=fva[0,s_-1]
            fvt[0,0]=l[i]; fvb[0,0]=h[i-2]; fva[0,0]=1
        if h[i]<l[i-2] and l[i-2]-h[i]>=0.15*a[i]:
            for s_ in range(3,0,-1): fvt[1,s_]=fvt[1,s_-1]; fvb[1,s_]=fvb[1,s_-1]; fva[1,s_]=fva[1,s_-1]
            fvt[1,0]=l[i-2]; fvb[1,0]=h[i]; fva[1,0]=1
        for s_ in range(4):  # إبطال بالإغلاق خلف المنطقة
            if oba[0,s_]==1 and c[i]<obb[0,s_]: oba[0,s_]=0
            if oba[1,s_]==1 and c[i]>obt[1,s_]: oba[1,s_]=0
            if fva[0,s_]==1 and c[i]<fvb[0,s_]: fva[0,s_]=0
            if fva[1,s_]==1 and c[i]>fvt[1,s_]: fva[1,s_]=0
        ai=a[i] if a[i]>0 else 1.0
        # فوليوم بروفايل 150 شمعة
        lo=1e18; hi=-1e18
        for q in range(i-150,i+1):
            lo=min(lo,l[q]); hi=max(hi,h[q])
        st=(hi-lo)/30.0+1e-12; w=np.zeros(30)
        for q in range(i-150,i+1):
            b0=min(29,max(0,int((l[q]-lo)/st))); b1=min(29,max(0,int((h[q]-lo)/st)))
            for b in range(b0,b1+1): w[b]+=v[q]/(b1-b0+1)
        pb=0
        for b in range(30):
            if w[b]>w[pb]: pb=b
        tot=w.sum(); acc=w[pb]; up=pb; dn=pb
        while acc<tot*0.7 and (up<29 or dn>0):
            nu=w[up+1] if up<29 else -1.0; nd=w[dn-1] if dn>0 else -1.0
            if nu>=nd: up+=1; acc+=w[up]
            else: dn-=1; acc+=w[dn]
        poc=lo+st*(pb+0.5); vah=lo+st*(up+1); val=lo+st*dn
        cb=min(29,max(0,int((c[i]-lo)/st)))
        h100=-1e18; l100=1e18
        for q in range(i-100,i+1):
            h100=max(h100,h[q]); l100=min(l100,l[q])
        for di in range(2):  # 0 شراء، 1 بيع
            d=1.0 if di==0 else -1.0
            # أقرب OB/FVG في اتجاه الصفقة (تحت السعر للشراء)
            bo=10.0; inob=0.0; bf=10.0; infv=0.0
            for s_ in range(4):
                if oba[di,s_]==1:
                    edge=obt[di,s_] if di==0 else obb[di,s_]
                    dist=(c[i]-edge)*d/ai
                    if c[i]<=obt[di,s_] and c[i]>=obb[di,s_]: inob=1.0; dist=0.0
                    if dist>=0 and dist<bo: bo=dist
                if fva[di,s_]==1:
                    edge=fvt[di,s_] if di==0 else fvb[di,s_]
                    dist=(c[i]-edge)*d/ai
                    if c[i]<=fvt[di,s_] and c[i]>=fvb[di,s_]: infv=1.0; dist=0.0
                    if dist>=0 and dist<bf: bf=dist
            fib=(h100-c[i])/(h100-l100+1e-12) if di==0 else (c[i]-l100)/(h100-l100+1e-12)
            edgeV=val if di==0 else vah
            F[i,di,0]=fib; F[i,di,1]=min(bo,10.0); F[i,di,2]=inob; F[i,di,3]=min(bf,10.0); F[i,di,4]=infv
            F[i,di,5]=(c[i]-poc)*d/ai; F[i,di,6]=1.0 if (c[i]>=val and c[i]<=vah) else 0.0
            F[i,di,7]=w[cb]/(w[pb]+1e-12); F[i,di,8]=(c[i]-edgeV)*d/ai

        eqhD=10.0; eqlD=10.0

        # بريميوم/ديسكاونت داخل آخر موجة كبيرة
