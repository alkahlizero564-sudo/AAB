import pandas as pd, numpy as np, warnings
warnings.filterwarnings('ignore')
from sklearn.linear_model import LogisticRegression
from sklearn.ensemble import RandomForestClassifier
from sklearn.preprocessing import StandardScaler
from sklearn.pipeline import make_pipeline
import lightgbm as lgb
S='/tmp/claude-0/-home-user-AAB/e2bd39c3-f0fb-5895-8329-383fc758e763/scratchpad'
df=pd.read_csv(S+'/ai/signals.csv')
meta=['market','tf','cat','ts','year','dir','win','r','src','stars','near','hot','held','st']
feat=[c for c in df.columns if c not in meta]
df['src_alka']=(df.src=='Alka')*1; df['src_aab']=(df.src=='AAB')*1; df['src_01']=(df.src=='0.1')*1
df['tfm']=df.tf.map({'15m':15,'1h':60,'4h':240})
feat+= ['src_alka','src_aab','src_01','tfm']
train=((df.market=='gold20')&(df.year>=2010)&(df.year<=2018)) | ((df.market=='btc')&(df.year<=2020))
test=~train & ~((df.market=='gold20')&(df.year<2010))
tr=df[train]; te=df[test]
print('train',len(tr),'win',tr.win.mean().round(3),' test',len(te),'win',te.win.mean().round(3))
X,y=tr[feat].values,tr.win.values
models={
 'logistic':make_pipeline(StandardScaler(),LogisticRegression(C=0.05,max_iter=2000)),
 'forest':RandomForestClassifier(n_estimators=400,min_samples_leaf=80,max_features=0.3,n_jobs=-1,random_state=1),
 'lightgbm':lgb.LGBMClassifier(n_estimators=300,learning_rate=0.02,num_leaves=15,min_child_samples=150,subsample=0.8,subsample_freq=1,colsample_bytree=0.6,reg_lambda=5,verbose=-1,random_state=1),
}
res={}
for name,m in models.items():
    m.fit(X,y)
    ptr=m.predict_proba(X)[:,1]; pte=m.predict_proba(te[feat].values)[:,1]
    q70=np.quantile(ptr,0.70); q85=np.quantile(ptr,0.85)
    te=te.assign(**{name:pte}); res[name]=(q70,q85)
    print(f"\n== {name}: train top30% win={tr.win[ptr>=q70].mean():.3f} top15%={tr.win[ptr>=q85].mean():.3f}")
def grp(row):
    m=row.market
    if m=='gold20': return 'ذهب 2019-25'
    if m=='btc': return 'بيتكوين 2021-26'
    if m=='eth': return 'إيثيريوم'
    if m.startswith(('eurusd','gbpusd','audusd','usdcad','usdjpy')): return 'عملات'
    if m.startswith(('nas','us30')): return 'مؤشرات أسهم'
    if m.startswith('oil'): return 'نفط'
    if m.startswith('silver'): return 'فضة'
    return m
te['grp']=te.apply(grp,axis=1)
rows=[]
for gname,g in list(te.groupby('grp'))+[('الكل',te)]:
    line=f"{gname:16s} n={len(g):5d} كل={g.win.mean()*100:5.1f}% | نجوم2+={g.win[g.stars>=2].mean()*100:5.1f}%({(g.stars>=2).sum()})"
    for name,(q70,q85) in res.items():
        a=g.win[g[name]>=q70]; b=g.win[g[name]>=q85]
        line+=f" | {name} أعلى30%={a.mean()*100:5.1f}%({len(a)}) أعلى15%={b.mean()*100:5.1f}%({len(b)})"
    print(line)
te.to_csv(S+'/ai/test_pred.csv',index=False)
