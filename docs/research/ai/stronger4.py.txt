import pandas as pd, numpy as np, warnings
warnings.filterwarnings('ignore')
from sklearn.ensemble import RandomForestClassifier
import lightgbm as lgb
S='/tmp/claude-0/-home-user-AAB/e2bd39c3-f0fb-5895-8329-383fc758e763/scratchpad'
df=pd.read_csv(S+'/ai/signals2.csv')
base=[c for c in df.columns if c not in ['market','tf','year','dir','win','r','src'] and not c.startswith('i_')]
indSafe=['i_st','i_stars','i_near','i_hot','i_held','i_sm','i_smstrong','i_km','i_daily200','i_sr_behind','i_sr_ahead','i_ribage','i_lt']
trm=((df.market=='gold20')&(df.year>=2010)&(df.year<=2018)) | ((df.market=='btc')&(df.year<=2020))
test=~trm & ~((df.market=='gold20')&(df.year<2010))
tr,te=df[trm],df[test]; gold=(te.market=='gold20').values
r=lambda a,ref: np.searchsorted(np.sort(ref),a)/len(ref)
def run(name,mk):
    v=[]
    for seed in range(1,6): v.append(mk(seed))
    out=f"{name:34s}"
    for lab,q in [('30%',.7),('15%',.85),('7%',.93)]:
        accs=[te.win[b>=np.quantile(a,q)].mean() for a,b in v]; ga=[te.win[(b>=np.quantile(a,q))&gold].mean() for a,b in v]
        out+=f" | {lab} {np.mean(accs)*100:5.1f}±{np.std(accs)*100:.1f} ذهب {np.mean(ga)*100:4.1f}"
    print(out)
def rfm(fs,n,depth,leaf):
    def f(seed):
        m=RandomForestClassifier(n_estimators=n,min_samples_leaf=leaf,max_depth=depth,max_features=0.3,n_jobs=-1,random_state=seed).fit(tr[fs],tr.win)
        return m.predict_proba(tr[fs])[:,1],m.predict_proba(te[fs])[:,1]
    return f
def lgbm(fs):
    def f(seed):
        m=lgb.LGBMClassifier(n_estimators=400,learning_rate=0.015,num_leaves=15,min_child_samples=200,subsample=0.8,subsample_freq=1,colsample_bytree=0.5,reg_lambda=10,verbose=-1,random_state=seed).fit(tr[fs],tr.win)
        return m.predict_proba(tr[fs])[:,1],m.predict_proba(te[fs])[:,1]
    return f
def both(fs,n,depth,leaf):
    def f(seed):
        a1,b1=rfm(fs,n,depth,leaf)(seed); a2,b2=lgbm(fs)(seed)
        return (r(a1,a1)+r(a2,a2))/2,(r(b1,a1)+r(b2,a2))/2
    return f
fs=base+indSafe
run('King (RF100 d8, 41)',rfm(base,100,8,150))
run('LGB (54 آمنة)',lgbm(fs))
run('RF150 d10 (54)',rfm(fs,150,10,120))
run('RF150 d10 + LGB (54)',both(fs,150,10,120))
run('RF100 d8 + LGB (54)',both(fs,100,8,150))
