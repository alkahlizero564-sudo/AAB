import pandas as pd, numpy as np, warnings
warnings.filterwarnings('ignore')
from sklearn.ensemble import RandomForestClassifier
S='/tmp/claude-0/-home-user-AAB/e2bd39c3-f0fb-5895-8329-383fc758e763/scratchpad'
df=pd.read_csv(S+'/ai/signals2.csv')
base=[c for c in df.columns if c not in ['market','tf','year','dir','win','r','src'] and not c.startswith('i_')]
safe=['i_st','i_stars','i_near','i_hot','i_held','i_sm','i_smstrong','i_km','i_daily200','i_sr_behind','i_sr_ahead','i_ribage','i_lt']

test=~trm & ~((df.market=='gold20')&(df.year<2010))
m=RandomForestClassifier(n_estimators=150,min_samples_leaf=120,max_depth=10,max_features=0.3,n_jobs=-1,random_state=1).fit(df[trm][base+safe].values.astype(np.float32),df.win[trm])
ptr=m.predict_proba(df[trm][base+safe].values.astype(np.float32))[:,1]
df['king']=m.predict_proba(df[base+safe].values.astype(np.float32))[:,1]
q={k:np.quantile(ptr,1-k/100) for k in [70,50,30]}
gold=df.market=='gold20'
