@njit(cache=True)
def hurst(lp,win,H):
    lags=np.array([1,2,4,8,16]); lx=np.log(lags.astype(np.float64))
    n=len(lp)
    for i in range(win+16,n):
        ys=np.zeros(5)
        for k in range(5):
            L=lags[k]; s=0.0; s2=0.0; m=0
            for t in range(i-win+L,i+1):
                dlt=lp[t]-lp[t-L]; s+=dlt; s2+=dlt*dlt; m+=1
            var=s2/m-(s/m)**2
            ys[k]=0.5*np.log(max(var,1e-20))
        mx=lx.mean(); my=ys.mean()
        H[i]=((lx-mx)*(ys-my)).sum()/((lx-mx)**2).sum()
