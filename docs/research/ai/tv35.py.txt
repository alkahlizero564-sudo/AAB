def simall(o,h,l,c,a,sig,w,r):

        ext=l[i]; 
        for j in range(i-2,i+1):
            if d>0: ext=min(ext,l[j])

        risk=(e-ext)*d+0.1*a[i]
        risk=min(max(risk,0.7*a[i]),2.5*a[i])
        t1=e+d*risk; t2=e+2*d*risk; st=e-d*risk; got=False; done=False
        for j in range(i+1,min(n,i+80)):

E=lambda x,n: x.ewm(span=n,adjust=False).mean()
def rsi(c,n):
    dlt=c.diff(); up=dlt.clip(lower=0).ewm(alpha=1/n,adjust=False).mean(); dn=(-dlt.clip(upper=0)).ewm(alpha=1/n,adjust=False).mean()
    return 100-100/(1+up/dn)
def atr(d,n=14):
    pc=d.close.shift(); tr=pd.concat([d.high-d.low,(d.high-pc).abs(),(d.low-pc).abs()],axis=1).max(axis=1)
    return tr.ewm(alpha=1/n,adjust=False).mean()
def cross(a,b): return ((a>b)&(a.shift()<=b.shift())).astype(int)-((a<b)&(a.shift()>=b.shift())).astype(int)
def flip(state): # state +1/-1 → إشارة عند التغيّر
    s=pd.Series(state); return ((s>0)&(s.shift()<=0)).astype(int)-((s<0)&(s.shift()>=0)).astype(int)
def trail_flip(c,a,mult):  # UT Bot / ATR trailing stop
    c=c.values; a=a.values; n=len(c); ts=np.zeros(n); st=np.zeros(n)
    for i in range(1,n):
        nl=a[i]*mult; p=ts[i-1]
        if c[i]>p and c[i-1]>p: ts[i]=max(p,c[i]-nl)
        elif c[i]<p and c[i-1]<p: ts[i]=min(p,c[i]+nl)
        elif c[i]>p: ts[i]=c[i]-nl
        else: ts[i]=c[i]+nl
        st[i]=1 if c[i]>ts[i] else -1
    return st
def supertrend(d,a,m=3):
    hl=(d.high+d.low)/2; up=(hl-m*a).values; dn=(hl+m*a).values; c=d.close.values; n=len(c)
    fu=up.copy(); fd=dn.copy(); tr=np.ones(n)
    for i in range(1,n):
        fu[i]=max(up[i],fu[i-1]) if c[i-1]>fu[i-1] else up[i]
        fd[i]=min(dn[i],fd[i-1]) if c[i-1]<fd[i-1] else dn[i]
        tr[i]=1 if c[i]>fd[i-1] else (-1 if c[i]<fu[i-1] else tr[i-1])
    return tr
def psar(d,step=0.02,mx=0.2):
    h=d.high.values; l=d.low.values; n=len(h); tr=np.ones(n); sar=l[0]; ep=h[0]; af=step; up=True
    for i in range(1,n):
        sar=sar+af*(ep-sar)
        if up:
            sar=min(sar,l[i-1],l[max(0,i-2)])
            if l[i]<sar: up=False; sar=ep; ep=l[i]; af=step
            elif h[i]>ep: ep=h[i]; af=min(af+step,mx)
        else:
            sar=max(sar,h[i-1],h[max(0,i-2)])
            if h[i]>sar: up=True; sar=ep; ep=h[i]; af=step
            elif l[i]<ep: ep=l[i]; af=min(af+step,mx)
        tr[i]=1 if up else -1
    return tr
