import pandas as pd, numpy as np, warnings
warnings.filterwarnings('ignore')
from sklearn.ensemble import RandomForestClassifier, ExtraTreesClassifier
from sklearn.linear_model import LogisticRegression
from sklearn.preprocessing import StandardScaler
from sklearn.pipeline import make_pipeline
import lightgbm as lgb
S='/tmp/claude-0/-home-user-AAB/e2bd39c3-f0fb-5895-8329-383fc758e763/scratchpad'
df=pd.read_csv(S+'/ai/signals2.csv')
base=[c for c in df.columns if c not in ['market','tf','year','dir','win','r','src'] and not c.startswith('i_')]
ind=[c for c in df.columns if c.startswith('i_')]
train=((df.market=='gold20')&(df.year>=2010)&(df.year<=2018)) | ((df.market=='btc')&(df.year<=2020))
test=~train & ~((df.market=='gold20')&(df.year<2010))
tr,te=df[train],df[test].copy()
def grp(mk): return {'gold20':'ذهب','btc':'بيتكوين','eth':'إيثيريوم'}.get(mk,'أخرى')
te['grp']=te.market.map(grp)
def rf(seed=1,leaf=150,depth=8,n=100): return RandomForestClassifier(n_estimators=n,min_samples_leaf=leaf,max_depth=depth,max_features=0.3,n_jobs=-1,random_state=seed)
def report(name,ptr,pte,extra=None):
    q70,q85,q93=np.quantile(ptr,.7),np.quantile(ptr,.85),np.quantile(ptr,.93)
    line=f"{name:40s}"
    for lab,q in [('30%',q70),('15%',q85),('7%',q93)]:
        sel=pte>=q
        if extra is not None: sel=sel&extra
        line+=f" | {lab} {te.win[sel].mean()*100:5.1f}%({sel.sum():5d})"
        g=te[sel&(te.grp=='ذهب')]; line+=f" ذهب {g.win.mean()*100:4.1f}"
    print(line)
# King
m=rf(); m.fit(tr[base],tr.win); kptr=m.predict_proba(tr[base])[:,1]; kte=m.predict_proba(te[base])[:,1]
report('King',kptr,kte)
near=te.entry_dist.values<0.75
report('King + دخول قريب (<0.75 ATR)',kptr,kte,near)
report('King + دخول قريب (<1.0 ATR)',kptr,kte,te.entry_dist.values<1.0)
# فرق التقدير
feats=base+ind
models={
 'rf':rf(1,120,10,300),
 'et':ExtraTreesClassifier(n_estimators=300,min_samples_leaf=100,max_depth=10,max_features=0.4,n_jobs=-1,random_state=1),
 'lgb':lgb.LGBMClassifier(n_estimators=400,learning_rate=0.015,num_leaves=15,min_child_samples=200,subsample=0.8,subsample_freq=1,colsample_bytree=0.5,reg_lambda=10,verbose=-1,random_state=1),
 'log':make_pipeline(StandardScaler(),LogisticRegression(C=0.03,max_iter=3000)),
}
P_tr,P_te={},{}
for k,mm in models.items():
    mm.fit(tr[feats],tr.win); P_tr[k]=mm.predict_proba(tr[feats])[:,1]; P_te[k]=mm.predict_proba(te[feats])[:,1]
    report(f'{k} (58)',P_tr[k],P_te[k])
# ترتيب بدل الاحتمالات للمجموعة
def rank(a,ref): return np.searchsorted(np.sort(ref),a)/len(ref)
ens_tr=np.mean([rank(P_tr[k],P_tr[k]) for k in models],axis=0)
ens_te=np.mean([rank(P_te[k],P_tr[k]) for k in models],axis=0)
report('مجموعة (rf+et+lgb+log)',ens_tr,ens_te)
report('مجموعة + دخول قريب <0.75',ens_tr,ens_te,near)
report('مجموعة + دخول قريب <1.0',ens_tr,ens_te,te.entry_dist.values<1.0)
ens2_tr=np.mean([rank(P_tr[k],P_tr[k]) for k in ['rf','et','lgb']],axis=0)
ens2_te=np.mean([rank(P_te[k],P_tr[k]) for k in ['rf','et','lgb']],axis=0)
report('مجموعة أشجار (rf+et+lgb)',ens2_tr,ens2_te)
report('مجموعة أشجار + قريب <0.75',ens2_tr,ens2_te,near)
