import pandas as pd, numpy as np, warnings
warnings.filterwarnings('ignore')
from sklearn.ensemble import RandomForestClassifier
S='/tmp/claude-0/-home-user-AAB/e2bd39c3-f0fb-5895-8329-383fc758e763/scratchpad'
df=pd.read_csv(S+'/ai/smc.csv')
meta=['market','tf','year','dir','win','r','vwin','vr']
feat=[c for c in df.columns if c not in meta]
df['tfm']=df.tf.map({'15m':15,'1h':60,'4h':240,'1d':1440}); feat.append('tfm')
trm=((df.market=='gold20')&(df.year>=2010)&(df.year<=2018)) | ((df.market=='btc')&(df.year<=2020))
test=~trm & ~((df.market=='gold20')&(df.year<2010))
def show(lab,m):
    print(f"{lab:38s} 1R: {df.win[m].mean()*100:5.1f}% avgR {df.r[m].mean():+.3f} | فيديو 2:1: {df.vwin[m].mean()*100:5.1f}% avgR {df.vr[m].mean():+.3f} | n={m.sum()}")
print('=== القاعدة وحدها')
show('تدريب', trm); show('اختبار', test)
for tf in ['15m','1h','4h','1d']: show(f'اختبار {tf}', test&(df.tf==tf))
show('ذهب اختبار', test&(df.market=='gold20')); show('بيتكوين اختبار', test&(df.market=='btc'))
show('ديسكاونت (<50%) اختبار', test&(df.disc<0.5)); show('ديسكاونت تدريب', trm&(df.disc<0.5))
tr,te=df[trm],df[test]
for tgt,lab in [('win','1R'),('vwin','فيديو 2:1')]:
    print(f'=== ذكاء اصطناعي (هدف {lab})')
    res=[]
    for seed in range(1,4):
        m=RandomForestClassifier(n_estimators=150,min_samples_leaf=120,max_depth=10,max_features=0.3,n_jobs=-1,random_state=seed).fit(tr[feat].values.astype(np.float32),tr[tgt])
        ptr=m.predict_proba(tr[feat].values.astype(np.float32))[:,1]; p=m.predict_proba(te[feat].values.astype(np.float32))[:,1]
        res.append((ptr,p))
        if seed==1:
            imp=sorted(zip(m.feature_importances_,feat),reverse=True)[:8]
    for top in [50,30,15]:
        a=[];ar=[];ga=[];ba=[];nn=[]
        for ptr,p in res:
            k=p>=np.quantile(ptr,1-top/100)
            a.append(te[tgt][k].mean()); ar.append(te['r' if tgt=='win' else 'vr'][k].mean()); nn.append(k.sum())
            ga.append(te[tgt][k&(te.market=='gold20').values].mean()); ba.append(te[tgt][k&(te.market=='btc').values].mean())
        print(f"  أعلى {top}%: {np.mean(a)*100:.1f}% avgR {np.mean(ar):+.3f} n={int(np.mean(nn))} | ذهب {np.mean(ga)*100:.1f}% | بيتكوين {np.mean(ba)*100:.1f}%")
    print('  أهم المعلومات:', ', '.join(f'{f}' for _,f in imp))
