@njit(cache=True)
def simall(o,h,l,c,a,sig,stp,w,r,w2,r2):
    n=len(c)
    for i in range(3,n-1):
        d=sig[i]
        if d==0 or a[i]<=0: continue
        e=o[i+1]
        ext=l[i] if d>0 else h[i]
        for j in range(i-2,i+1):
            ext=min(ext,l[j]) if d>0 else max(ext,h[j])
        risk=min(max((e-ext)*d+0.1*a[i],0.7*a[i]),2.5*a[i])
        t1=e+d*risk; t2=e+2*d*risk; st=e-d*risk; got=False; done=False
        for j in range(i+1,min(n,i+80)):
            if (d>0 and l[j]<=st) or (d<0 and h[j]>=st):
                w[i]=1 if got else 0; r[i]=0.5 if got else -1.0; done=True; break
            if (not got) and ((d>0 and h[j]>=t1) or (d<0 and l[j]<=t1)): got=True; st=e
            if got and ((d>0 and h[j]>=t2) or (d<0 and l[j]<=t2)): w[i]=1; r[i]=1.5; done=True; break
        if not done: w[i]=1 if got else 0; r[i]=0.5 if got else -1.0
        # 2:1 بوقف الاستراتيجية (إن وُجد) وإلا خلف 3 شموع
        sp=stp[i] if not np.isnan(stp[i]) else ext-d*0.1*a[i]
        rk=(e-sp)*d
        if rk<0.2*a[i]: continue
        tp=e+2*d*rk; w2[i]=0; r2[i]=-1.0
        for j in range(i+1,min(n,i+300)):
            if (d>0 and l[j]<=sp) or (d<0 and h[j]>=sp): break
            if j>i+1 and ((d>0 and h[j]>=tp) or (d<0 and l[j]<=tp)): w2[i]=1; r2[i]=2.0; break

    d['time']=pd.to_datetime(d.time,utc=True,errors='coerce'); return d.dropna(subset=['time']).reset_index(drop=True)

    delta=(2*d.takerbuy.values-v) if mode=='real' else v*(c-o)/rg

    obs=[]  # (known_i, dir, top, bot, used, broken)

        if lastH is not None and not hB and c[i]>h[lastH]:

                touched=(l[i]<=edge) if dd>0 else (h[i]>=edge)
                if touched and hv and (c[i]-edge)*dd>0 and (c[i]-o[i])*dd>0:
                    sig['OB بحجم عالٍ (ارتداد)'][i]=dd; stp['OB بحجم عالٍ (ارتداد)'][i]=far-dd*0.1*a[i]; z[4]=True
            else:
                # Breaker: رجوع للمنطقة من الجهة الأخرى ورفض → صفقة عكس اتجاهها الأصلي
                nd=-dd; e2=far
