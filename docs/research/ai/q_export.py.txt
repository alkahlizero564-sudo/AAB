def flat(booster):
    dm=booster.dump_model(); F=[];T=[];L=[];R=[];V=[];starts=[]
    for t in dm['tree_info']:
        starts.append(len(F))
        def rec(nd):
            idx=len(F); F.append(-1);T.append(0.0);L.append(-1);R.append(-1);V.append(0.0)
            if 'split_index' in nd:
                assert nd['decision_type']=='<='
                F[idx]=nd['split_feature']; T[idx]=nd['threshold']
                l=rec(nd['left_child']); r=rec(nd['right_child']); L[idx]=l; R[idx]=r
            else: V[idx]=nd['leaf_value']
            return idx
        rec(t['tree_structure'])
    return starts,F,T,L,R,V
def pred(fl,x):
    st,F,T,L,R,V=fl; s=0.0
    for o in st:
        n=o
        while F[n]!=-1: n=L[n] if x[F[n]]<=T[n] else R[n]
        s+=V[n]
    return s
