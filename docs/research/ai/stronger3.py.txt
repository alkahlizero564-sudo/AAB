import pandas as pd, numpy as np, warnings
warnings.filterwarnings('ignore')
from sklearn.ensemble import RandomForestClassifier
import lightgbm as lgb
S='/tmp/claude-0/-home-user-AAB/e2bd39c3-f0fb-5895-8329-383fc758e763/scratchpad'
df=pd.read_csv(S+'/ai/signals2.csv')
base=[c for c in df.columns if c not in ['market','tf','year','dir','win','r','src'] and not c.startswith('i_')]
ind=[c for c in df.columns if c.startswith('i_')]
trm=((df.market=='gold20')&(df.year>=2010)&(df.year<=2018)) | ((df.market=='btc')&(df.year<=2020))
test=~trm & ~((df.market=='gold20')&(df.year<2010))
tr,te=df[trm],df[test]
gold=(te.market=='gold20').values
r=lambda a,ref: np.searchsorted(np.sort(ref),a)/len(ref)
res={'King':[], 'جديد (RF+LGB + مؤشراتنا)':[]}
for seed in range(1,6):
    m=RandomForestClassifier(n_estimators=100,min_samples_leaf=150,max_depth=8,max_features=0.3,n_jobs=-1,random_state=seed).fit(tr[base],tr.win)
    a,b=m.predict_proba(tr[base])[:,1],m.predict_proba(te[base])[:,1]
    res['King'].append((a,b))
    fs=base+ind
    m1=RandomForestClassifier(n_estimators=300,min_samples_leaf=120,max_depth=10,max_features=0.3,n_jobs=-1,random_state=seed).fit(tr[fs],tr.win)
    m2=lgb.LGBMClassifier(n_estimators=400,learning_rate=0.015,num_leaves=15,min_child_samples=200,subsample=0.8,subsample_freq=1,colsample_bytree=0.5,reg_lambda=10,verbose=-1,random_state=seed).fit(tr[fs],tr.win)
    p1,p2=m1.predict_proba(tr[fs])[:,1],m2.predict_proba(tr[fs])[:,1]; q1,q2=m1.predict_proba(te[fs])[:,1],m2.predict_proba(te[fs])[:,1]
    res['جديد (RF+LGB + مؤشراتنا)'].append(((r(p1,p1)+r(p2,p2))/2,(r(q1,p1)+r(q2,p2))/2))
for k,v in res.items():
    out=f"{k:28s}"
    for lab,q in [('30%',.7),('15%',.85),('7%',.93)]:
        accs=[te.win[b>=np.quantile(a,q)].mean() for a,b in v]; ga=[te.win[(b>=np.quantile(a,q))&gold].mean() for a,b in v]
        out+=f" | {lab} {np.mean(accs)*100:5.1f}±{np.std(accs)*100:.1f} ذهب {np.mean(ga)*100:4.1f}"
    print(out)
